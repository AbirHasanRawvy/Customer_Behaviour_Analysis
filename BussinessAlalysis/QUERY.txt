SELECT
    gender,
    SUM(`purchase_amount(usd)`) AS revenue
FROM customer
GROUP BY gender;

select category, sum('purchase_amount(usd)') as revenue, count(*) as total_sales from customer 
group by category
order by revenue;



select * from customer limit 5;
# Which season the revenue will earn most?
select season, sum('purchase_amount(usd)') as revenue, count(*) as total_sales from customer
group by season
order by revenue desc;

#Which payment methods were most used?
select payment_method, sum('purchase_amount(usd)') as Revenue, count(*) as total_sales from customer
group by payment_method
order by revenue desc;

#Which age group of company work most or earn most amount of revenue?
select age, sum('purchase_amount(usd)') as revenue, count(*) as total_sales from customer
group by age
order by revenue desc;

# If you give a discount for customer, the revenue will increase or decrease?
